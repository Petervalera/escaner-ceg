async function alDetectarQR(codigoCedula) { html5QrCode.pause(true); mostrarEstado("Cargando...", "Buscando en Supabase...", "🟡"); const cedulaLeida = codigoCedula.trim(); const soloNum = cedulaLeida.replace(/\D/g, ''); const conV = `V-${soloNum}`; console.log("Buscando cédula:", cedulaLeida); try { // 1. Buscar en registro_general asegurando la coincidencia con la columna 'cedula' const { data: usuario, error: errUser } = await _supabase .from('registro_general') .select('*') .or(`cedula.eq.${conV},cedula.eq.${soloNum},cedula.eq.${cedulaLeida}`) .maybeSingle(); if (errUser) { console.error("Error consultando usuario:", errUser); mostrarEstado("error", `Error BD: ${errUser.message}`, "🔴"); reagendarReinicio(); return; } if (!usuario) { console.warn("No se encontró el usuario con cédula:", cedulaLeida); mostrarEstado("no_encontrado", `Cédula ${cedulaLeida} no registrada.`, "🔴"); reagendarReinicio(); return; } console.log("Usuario encontrado:", usuario.nombre_apellido); // 2. Determinar Turno y Semáforo (Entrada / Salida) const ahora = new Date(); const horaDecimal = ahora.getHours() + (ahora.getMinutes() / 60); let estadoSemaforo = "🟢 A TIEMPO"; let tipoMovimiento = "ENTRADA"; if (horaDecimal >= 11.5 && horaDecimal <= 13.0) { tipoMovimiento = "SALIDA"; } else if (horaDecimal > 17.5) { tipoMovimiento = "SALIDA"; } if (tipoMovimiento === "ENTRADA") { if (horaDecimal > 7.25 && horaDecimal <= 7.5) { estadoSemaforo = "🟡 RETARDO"; } else if (horaDecimal > 7.5) { estadoSemaforo = "🔴 LLEGADA TARDIA"; } } // 3. Registrar Asistencia en la tabla 'asistencia' const payloadAsistencia = { nombre_apellido: usuario.nombre_apellido || "Sin Nombre", cedula: usuario.cedula || conV, tipo_movimiento: tipoMovimiento, estado_asistencia: estadoSemaforo, fecha_hora: ahora.toISOString() }; const { error: errAsist } = await _supabase .from('asistencia') .insert([payloadAsistencia]); if (errAsist) { console.error("Error al insertar asistencia:", errAsist); mostrarEstado("error", `Error Asistencia: ${errAsist.message}`, "🔴"); reagendarReinicio(); return; } // 4. Notificación por Telegram (Controlada para que no bloquee si falta el teléfono) try { if (usuario.telefono && typeof enviarTelegram === 'function') { enviarTelegram(usuario.telefono, usuario.nombre_apellido, tipoMovimiento, estadoSemaforo); } } catch (errTel) { console.warn("Aviso Telegram omitido:", errTel); } // 5. Mostrar éxito en pantalla con el estado actual mostrarEstado("exito", `${usuario.nombre_apellido}
${tipoMovimiento} - ${estadoSemaforo}`, "🟢"); } catch (err) { console.error("Excepción en escaneo:", err); mostrarEstado("error", "Error procesando acceso.", "🔴"); } reagendarReinicio(); }
