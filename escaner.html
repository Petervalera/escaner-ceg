async function alDetectarQR(codigoCedula) { html5QrCode.pause(true); mostrarEstado("Cargando...", "Buscando datos...", "🟡"); // Limpiar texto leído del QR const cedulaLimpia = codigoCedula.trim(); const soloNum = cedulaLimpia.replace(/\D/g, ''); // Extrae solo números (34098890) const conV = `V-${soloNum}`; // Formato con V- (V-34098890) try { // 1. Buscar en registro_general probando con V-, sin V- o texto exacto const { data: usuario, error: errUser } = await _supabase .from('registro_general') .select('*') .or(`cedula.eq.${conV},cedula.eq.${soloNum},cedula.eq.${cedulaLimpia}`) .maybeSingle(); if (errUser) { mostrarEstado("error", `Error BD Usuario: ${errUser.message}`, "🔴"); reagendarReinicio(); return; } if (!usuario) { mostrarEstado("no_encontrado", `Cédula ${cedulaLimpia} NO registrada`, "🔴"); reagendarReinicio(); return; } // 2. Determinar Turno y Semáforo const ahora = new Date(); const horaDecimal = ahora.getHours() + (ahora.getMinutes() / 60); let estadoSemaforo = "🟢 A TIEMPO"; let tipoMovimiento = "ENTRADA"; if (horaDecimal >= 11.5 && horaDecimal <= 13.0) { tipoMovimiento = "SALIDA"; } else if (horaDecimal > 17.5) { tipoMovimiento = "SALIDA"; } if (tipoMovimiento === "ENTRADA") { if (horaDecimal > 7.25 && horaDecimal <= 7.5) { estadoSemaforo = "🟡 RETARDO"; } else if (horaDecimal > 7.5) { estadoSemaforo = "🔴 LLEGADA TARDIA"; } } // 3. Registrar Asistencia en Supabase const { error: errAsist } = await _supabase .from('asistencia') .insert([{ nombre_apellido: usuario.nombre_apellido || "Sin Nombre", cedula: usuario.cedula || conV, tipo_movimiento: tipoMovimiento, estado_asistencia: estadoSemaforo, fecha_hora: ahora.toISOString() }]); if (errAsist) { console.error("Error asistencia:", errAsist); mostrarEstado("error", `Error guardar: ${errAsist.message}`, "🔴"); reagendarReinicio(); return; } // 4. Intento seguro de Telegram (silencioso si falla) try { if (usuario.telefono) { enviarTelegram(usuario.telefono, usuario.nombre_apellido, tipoMovimiento, estadoSemaforo); } } catch (eTelegram) { console.warn("Error en Telegram ignorado:", eTelegram); } mostrarEstado("exito", `${usuario.nombre_apellido}
${tipoMovimiento} - ${estadoSemaforo}`, "🟢"); } catch (err) { console.error("Error general:", err); mostrarEstado("error", "Error inesperado procesando.", "🔴"); } reagendarReinicio(); }
